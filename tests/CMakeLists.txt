cmake_minimum_required(VERSION 3.16)
project(rf1_audio_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
find_path(OPUS_INCLUDE_DIR opus.h PATH_SUFFIXES opus)
find_library(OPUS_LIBRARY NAMES opus libopus.so.0)
if(NOT OPUS_INCLUDE_DIR OR NOT OPUS_LIBRARY)
    message(FATAL_ERROR "Install libopus-dev, or set OPUS_INCLUDE_DIR and OPUS_LIBRARY to a host Opus build")
endif()
add_executable(live_audio_tests live_audio_tests.cpp ../src/logic/live_audio_runtime.cpp)
target_include_directories(live_audio_tests PRIVATE stubs ../src "${OPUS_INCLUDE_DIR}")
target_compile_features(live_audio_tests PRIVATE cxx_std_17)
target_compile_options(live_audio_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(live_audio_tests PRIVATE "${OPUS_LIBRARY}" Threads::Threads)
add_test(NAME bounded_live_audio COMMAND live_audio_tests)
set_tests_properties(bounded_live_audio PROPERTIES TIMEOUT 20)
